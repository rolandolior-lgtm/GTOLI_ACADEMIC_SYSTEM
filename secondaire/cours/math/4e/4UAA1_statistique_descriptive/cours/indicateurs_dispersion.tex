\section{Indicateurs de dispersion}

\begin{cadretheorie}
Les indicateurs de dispersion permettent d’évaluer si les données sont proches ou éloignées les unes des autres.

Les principaux indicateurs sont :
\begin{itemize}
    \item l’étendue ;
    \item la variance ;
    \item l’écart-type ;
    \item l’intervalle interquartile.
\end{itemize}
\end{cadretheorie}

\begin{cadrevert}{Formules}
\[
\text{Étendue}=x_{\max}-x_{\min}
\]

\[
s^2=\frac{1}{n}\sum_{i=1}^{n}(x_i-\bar{x})^2
\]

\[
s=\sqrt{s^2}
\]
\end{cadrevert}
